#include <bits/stdc++.h>

using namespace std;

#ifdef LOCAL
#include "algo/debug.h"
#else
#define debug(...) 42
#endif

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int tt;
  cin >> tt;
  for (int qq = 1; qq <= tt; qq++) {
    cout << "Case #" << qq << ": ";
    int n, m;
    cin >> n >> m;
    vector<int> a(n);
    for (int i = 0; i < n; i++) {
      cin >> a[i];
    }
    vector<int> b(m);
    for (int i = 0; i < m; i++) {
      cin >> b[i];
    }
    sort(a.rbegin(), a.rend());
    vector<int> ac;
    for (int i = 0; i < n; i++) {
      if (i == 0 || a[i] != a[i - 1]) {
        ac.push_back(1);
      } else {
        ac.back() += 1;
      }
    }
    sort(b.rbegin(), b.rend());
    int low = 0, high = int(ac.size()) + 1;
    while (high - low > 1) {
      int mid = (low + high) >> 1;
      bool ok = true;
      int ppl = 0;
      int64_t need_items = 0;
      int64_t got_items = 0;
      int ptr = m;
      for (int i = 0; i < mid; i++) {
        for (int j = 0; j < ac[i]; j++) {
          ppl += 1;
          need_items += mid - i;
          while (ptr > 0 && b[ptr - 1] < ppl) {
            ptr -= 1;
          }
          got_items += ptr;
          ok &= (got_items >= need_items);
        }
      }
      if (ok) {
        low = mid;
      } else {
        high = mid;
      }
    }
    cout << accumulate(ac.begin(), ac.begin() + low, 0) << '\n';    
  }
  return 0;
}
